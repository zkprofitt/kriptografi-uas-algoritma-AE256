-- 1. Buat database
CREATE DATABASE IF NOT EXISTS db_fintech;
USE db_fintech;

-- 2. Buat tabel utama
CREATE TABLE pengajuan_pinjol (
    id                   INT AUTO_INCREMENT PRIMARY KEY,

    -- Label display (plaintext, untuk identifikasi di Panel Admin)
    nama_display         VARCHAR(255),

    -- Data Terenkripsi (AES-256-CBC, disimpan dalam format Base64)
    nama_lengkap         TEXT,
    pekerjaan            TEXT,
    perusahaan           TEXT,
    tenor                TEXT,
    tujuan_pinjaman      TEXT,
    status_nikah         TEXT,
    nama_bank_user_enc   TEXT,
    nik_enc              TEXT,
    alamat_enc           TEXT,
    email_enc            TEXT,
    gaji_enc             TEXT,
    hp_enc               TEXT,
    kontak_darurat_enc   TEXT,
    rekening_enc         TEXT,
    nominal_pinjaman_enc TEXT,

    -- IV unik per baris (Base64), digunakan untuk proses dekripsi
    iv_data              VARCHAR(255),

    -- Timestamp otomatis saat data dimasukkan
    tgl_pengajuan        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
