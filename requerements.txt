-- ============================================================
--  Safe-Loan: Database Setup
--  Keamanan: AES-256-CBC + RSA-2048 + HMAC-SHA512
--
--  Catatan format kolom enkripsi:
--  Setiap kolom _enc menyimpan 3 komponen digabung dengan separator "||":
--  FORMAT: <ciphertext_base64>||<encrypted_aes_key_base64>||<hmac_sha512_base64>
--
--  Jalankan file ini di MySQL/phpMyAdmin sebelum menjalankan app.
-- ============================================================

CREATE DATABASE IF NOT EXISTS db_fintech;
USE db_fintech;

CREATE TABLE pengajuan_pinjol (
    id                   INT AUTO_INCREMENT PRIMARY KEY,

    -- Label display (plaintext, untuk identifikasi di Panel Admin)
    nama_display         VARCHAR(255),

    -- Data Terenkripsi AES-256-CBC
    -- Format tiap kolom: ciphertext||encrypted_aes_key||hmac_sha512 (Base64, separator ||)
    nama_lengkap         TEXT,
    pekerjaan            TEXT,
    perusahaan           TEXT,
    tenor                TEXT,
    tujuan_pinjaman      TEXT,
    status_nikah         TEXT,
    nama_bank_user_enc   TEXT,
    nik_enc              TEXT,
    alamat_enc           TEXT,
    email_enc            TEXT,
    gaji_enc             TEXT,
    hp_enc               TEXT,
    kontak_darurat_enc   TEXT,
    rekening_enc         TEXT,
    nominal_pinjaman_enc TEXT,

    -- IV bersama per transaksi (Base64), digunakan untuk proses dekripsi
    iv_data              VARCHAR(255),

    -- Timestamp otomatis saat data dimasukkan
    tgl_pengajuan        TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
